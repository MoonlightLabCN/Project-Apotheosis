# Apotheosis: WebCore platform glue for the WinUWP port (ARM32 App Container).
#
# Modeled on Source/WebCore/PlatformWin.cmake, stripped to the App-Container-safe
# subset and re-pointed at the Cairo software graphics backend plus the
# FreeType/Fontconfig/HarfBuzz font backend.
#
# Rules applied against PlatformWin.cmake:
#   * KEEP    compiler-only flags; the coordinate/struct conversion *Win.cpp files;
#             Curl/OpenSSL/ImageDecoders/Cairo glue; portable text and locale code;
#             read-only Win helpers (SystemInfo/UserAgent/MIME/Logging).
#   * DROP    MSAA/oleacc accessibility; clipboard/drag/OLE pasteboard; the native
#             <select> popup; CreateWindow fullscreen and overlays; cursors;
#             GDI image/icon/DIB painting; MediaFoundation video; SharedMemoryWin;
#             Uniscribe and the win/win-cairo GDI+DirectWrite font backend.
#   * REPLACE the font block with platform/FreeType.cmake.
#   * DROP    the usp10 link library (Uniscribe -> HarfBuzz) and the whole
#             MediaFoundation INTERFACE library (ENABLE_VIDEO and
#             USE_MEDIA_FOUNDATION are forced off, so that block never runs).
#
# Files marked "STUB:" compile but call desktop-only APIs; they need a
# NotImplemented or WinRT replacement before the driver links and runs. That
# link error surface is milestone M1c, not this file's problem.

add_definitions(/bigobj -D__STDC_CONSTANT_MACROS)

# --- Form-control theming ---------------------------------------------------
# Adwaita paints controls purely through Cairo (no UxTheme/GDI), so it is the
# App-Container-safe RenderTheme. We deliberately do not take the RenderThemeWin
# path.
include(platform/Adwaita.cmake)

# --- Networking, crypto, image decoding -------------------------------------
include(platform/Curl.cmake)
include(platform/ImageDecoders.cmake)
include(platform/OpenSSL.cmake)

# --- Graphics backend: Cairo software image surfaces ------------------------
# Cairo.cmake's unified list still contains GraphicsContextGLCairo.cpp, but that
# file is self-guarded by '#if ENABLE(WEBGL) && USE(CAIRO)' and compiles to
# nothing here. The GdkPixbuf branch in ImageBufferUtilitiesCairo.cpp is inert
# because PLATFORM(GTK) is undefined for PORT=WinUWP.
if (USE_CAIRO)
    include(platform/Cairo.cmake)
endif ()

# --- Font/text backend: FreeType + Fontconfig + HarfBuzz --------------------
# This replaces the entire win/win-cairo GDI + DirectWrite + Uniscribe + MLang
# font stack and SystemFontDatabaseWin. FreeType.cmake also pulls in
# platform/graphics/cairo/FontCairoHarfbuzzNG.cpp, which is the Cairo glyph-draw
# glue the dropped Font*WinCairo files used to provide.
# UWP-TODO (runtime, not cmake): Fontconfig needs a writable cache directory or a
# baked-in fonts.conf inside the App Container sandbox.
include(platform/FreeType.cmake)

# The non-font survivors of PlatformWin.cmake's USE_CAIRO block are dropped
# after all: both files are pure GDI interop (cairo_win32_surface from HDC /
# HBITMAP) and require the cairo-win32 backend, which cannot exist in an App
# Container - our cairo is built without it (see local patch skipping the GDI
# win32 surface/font backends under WINAPI_FAMILY_APP). If the driver link
# misses GraphicsContext::getWindowsContext/releaseWindowsContext or
# ImageAdapter symbols, stub them NotImplemented in the driver instead of
# reintroducing GDI.
